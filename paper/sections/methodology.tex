\section{The \fedhep{} Framework}
\label{sec:method}

\paragraph{Problem Setting and Threat Model.}
We consider a distributed federated multiagent network of $N$ heterogeneous edge clients $\mathcal{N} = \{1, \dots, N\}$. Each client $i$ possesses a private local dataset $\mathcal{D}_i = \{(x_k, y_k)\}_{k=1}^{n_i}$ sampled from distribution $\mathcal{P}_i(x, y)$ over input space $\mathcal{X}$ and label space $\mathcal{Y} = \{1, \dots, C\}$. The client's observed label support is denoted by $\mathcal{Y}_i = \{y \in \mathcal{Y} \mid \exists (x, y) \in \mathcal{D}_i\}$. Under severe Dirichlet partitioning, $|\mathcal{Y}_i| \ll C$. We assume an omniscient Byzantine adversary controlling up to a fraction $q < 0.5$ of clients $\mathcal{M} \subset \mathcal{N}$ executing coordinated attacks including label flipping ($y \to C - 1 - y$), sign flipping ($\Delta \mathbf{w} \to -\gamma \Delta \mathbf{w}$), gradient ascent, and Gaussian perturbation~\cite{fang2020local}. The server maintains no clean proxy dataset and cannot inspect local telemetry.

\subsection{Hierarchical Architecture and Learning Mechanism}
\label{sec:architecture}

\paragraph{Single-Backbone Three-Tier Heads.}
Unlike dual-model paradigms that duplicate whole networks~\cite{li2021ditto}, \fedhep{} maintains a single shared convolutional feature extractor $\Phi_\theta: \mathcal{X} \to \mathbb{R}^d$ parameterized by $\theta$ (Figure~\ref{fig:architecture}). Branched from this shared backbone are three linear classification heads $W \in \mathbb{R}^{d \times C}$:
(1) \textbf{Root Head ($W_r$)}, aggregated globally across all $N$ clients via Byzantine-resilient consensus for generalized invariant features;
(2) \textbf{Parent Head ($W_p$)}, synchronized exclusively within peer clusters $k \in \{1, \dots, K\}$ to pool complementary label representations; and
(3) \textbf{Local Head ($W_l$)}, trained strictly on-device without communication for local specialization.

During on-device inference, predictions are generated via an anchored linear combination of temperature-calibrated logits~\cite{guo2017calibration}:
\begin{equation}
\label{eq:ensemble}
\mathbf{z}_{\mathrm{ens}} = w_l \frac{\mathbf{z}_{i,l}}{T_l} + w_p \frac{\mathbf{z}_{i,p}}{T_p} + w_r \frac{\mathbf{z}_{i,r}}{T_r}, \quad \hat{y} = \argmax_{c \in \mathcal{Y}} z_{\mathrm{ens}, c},
\end{equation}
where $\mathbf{z}_{i,m} = W_m^\top \Phi_\theta(x)$ and $T_m$ denote head-specific calibration temperatures, preserving predictive sharpness without softmax probability dilution.

\paragraph{Active-Class Logit Masking (ACLM).}
In decoupled architectures under label skew, evaluating standard cross-entropy over unobserved categories forces unobserved logits toward $-\infty$, injecting damaging negative gradients that pull shared backbone features $\theta$ away from valid geometric representations. To eliminate this interference, \fedhep{} introduces \textbf{ACLM}. For client $i$ with observed support indicator $M_{i,c} = \mathbb{I}(c \in \mathcal{Y}_i)$, logits are masked prior to softmax:
\begin{equation}
\label{eq:aclm}
\tilde{z}_{i,c} = 
\begin{cases} 
z_{i,c}, & \text{if } c \in \mathcal{Y}_i, \\ 
-\infty, & \text{if } c \notin \mathcal{Y}_i.
\end{cases}
\end{equation}
Confining the active logit support ensures that $\frac{\partial \mathcal{L}_{\mathrm{CE}}}{\partial z_{i,c}} = 0$ for all $c \notin \mathcal{Y}_i$, shielding shared feature representations $\Phi_\theta$ from unobserved category penalties.

\paragraph{Skew-Calibrated Loss Weighting.}
To smoothly balance global consensus and local personalization, each client computes its normalized empirical label entropy $H(\mathbf{p}_i) = -\sum_{c \in \mathcal{Y}_i} p_{i,c} \ln p_{i,c}$, defining skew coefficient:
\begin{equation}
\label{eq:rskew}
\Rskew^{(i)} = 1 - \frac{H(\mathbf{p}_i)}{\ln C} \in [0, 1].
\end{equation}
Under uniform data, $\Rskew \to 0$ (IID); under acute one-class concentration, $\Rskew \to 1$. We enforce a data-support floor anchor $a_i = \max\left(\frac{1}{2K}, \frac{|\mathcal{Y}_i|}{C}\right)$ and compute unnormalized Bernstein polynomial weights:
\begin{equation}
\label{eq:binomial_weights}
\lambda_r = a_i + (1 - a_i)(\Rskew^{(i)})^2, \quad \lambda_p = 2\Rskew^{(i)}(1 - \Rskew^{(i)}), \quad \lambda_l = (1 - \Rskew^{(i)})^2.
\end{equation}
The exact partition-of-unity coefficients are obtained via $\alpha_k = \frac{\lambda_k}{\sum_{j \in \{r, p, l\}} \lambda_j}$, forming the composite objective:
\begin{equation}
\mathcal{L}_{\mathrm{total}}(\theta, W_r, W_p, W_l) = \alpha_r \mathcal{L}_r(W_r; \theta) + \alpha_p \mathcal{L}_p^{\mathrm{mask}}(W_p; \theta) + \alpha_l \mathcal{L}_l^{\mathrm{mask}}(W_l; \theta).
\end{equation}

\paragraph{Privacy-Preserving Collaborative Clustering via Random Projections.}
To group clients into collaborative peer clusters without exposing raw local parameters, clients project Parent Head updates $\Delta W_{p,i} \in \mathbb{R}^{d \cdot C}$ into a compact subspace using a shared Gaussian projection matrix $\mathbf{R} \in \mathbb{R}^{m \times D}$ ($\mathbf{R}_{jk} \sim \mathcal{N}(0, 1/m), m=256$), followed by $L_2$ normalization:
\begin{equation}
\label{eq:sketch}
\mathbf{s}_i = \frac{\mathbf{R} \cdot \mathrm{vec}(\Delta W_{p,i})}{\|\mathbf{R} \cdot \mathrm{vec}(\Delta W_{p,i})\|_2} \in \mathbb{R}^m.
\end{equation}
By the Johnson-Lindenstrauss lemma~\cite{johnson1984extensions}, pairwise Euclidean distances between normalized sketches accurately approximate pairwise cosine geometries within distortion $\epsilon$ whenever $m \ge \mathcal{O}(\epsilon^{-2} \log N)$. Because $m=256 \ll D \approx 10^5$, update reconstruction is mathematically underdetermined, preserving edge privacy while enabling the server to partition clients into $K$ synergistic clusters.

\subsection{Layered Subspace Byzantine Defense}
\label{sec:defense}

To defend against malicious poisoning without penalizing legitimate non-IID angular deviations, \fedhep{} implements a three-tier defense decoupling head verification from backbone aggregation:
\begin{itemize}
    \item \textbf{Subspace-Constrained Cosine Filtering (SCCF)}: Conventional Euclidean aggregators fail under non-IID distributions because honest clients with disjoint classes exhibit large natural angular deviations. For each client $i$, the server projects head update deltas $\Delta \mathbf{w}_i$ onto the subspace $\mathcal{S}_i$ spanned by client $i$'s observed coordinates:
    \begin{equation}
    \label{eq:sccf}
    \tau_i^{(t)} = \Softmax\left( \frac{\cos\left(\mathbf{P}_{\mathcal{S}_i} \Delta \mathbf{w}_i, \mathbf{P}_{\mathcal{S}_i} \mathbf{c}^{(t)}\right)}{T_{\mathrm{def}}} \right), \quad \mathcal{S}_i = \operatorname{span}\{\mathbf{e}_k : k \in \coords(\mathcal{Y}_i)\},
    \end{equation}
    where $\mathbf{c}^{(t)}$ is the consensus direction and $T_{\mathrm{def}}$ is the defense temperature. By restricting cosine similarity checks strictly to observed active class subspaces $\mathcal{S}_i$, SCCF eliminates false-positive penalization of honest specialized clients.
    \item \textbf{Adaptive Backbone Norm-Bounding}: To neutralize unbounded scaling attacks on shared features, backbone updates $\Delta \theta_i$ are constrained by an adaptive clipping threshold $\tau$:
    \begin{equation}
    \tilde{\Delta} \theta_i = \Delta \theta_i \cdot \min\left(1, \frac{\tau}{\|\Delta \theta_i\|_2}\right).
    \end{equation}
    \item \textbf{Temporal Trust Tracking (TTT)}: To suppress intermittent or alternating adversaries, the server updates an exponential moving average trust score $T_i^{(t)} \in [0, 1]$:
    \begin{equation}
    T_i^{(t)} = \beta T_i^{(t-1)} + (1 - \beta) \cdot \min\left(1.0, |\mathcal{C}_{k(i)}| \cdot \tau_i^{(t)}\right),
    \end{equation}
    where $\mathcal{C}_{k(i)}$ is the peer cluster of client $i$. Clients whose trust decays below threshold $\theta_{\mathrm{iso}}$ are excluded from aggregation, neutralizing stealthy Byzantine drift over multi-round horizons.
\end{itemize}
